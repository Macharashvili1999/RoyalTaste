ინსტრუქცია: როგორ ატვირთოთ GitHub-ზე
  1. შედით GitHub.com-ზე და შექმენით ახალი რეპოზიტორი (New Repository).
  2. დაარქვით სახელი (მაგალითად: catering-site) და დააჭირეთ Create repository.
  3. დააჭირეთ Creating a new file, ფაილის სახელში ჩაწერეთ index.html.
  4. ჩასვით ქვემოთ მოცემული სრული HTML კოდი და დააჭირეთ Commit changes.
  5. გადადით Settings -> Pages, Source-ში აირჩიეთ main branch და დააჭირეთ Save.

        let cart = {};

        function renderProducts(filter = 'all') {
            const grid = document.getElementById('product-grid');
            const items = filter === 'all' ? products : products.filter(p => p.cat === filter);
            grid.innerHTML = items.map(p => `
                <div class="card">
                    <img src="${p.img}" alt="${p.name}">
                    <div class="card-body">
                        <div class="card-title">${p.name}</div>
                        <div class="card-desc">${p.desc}</div>
                        <div class="card-footer">
                            <div class="price">${p.price} €</div>
                            <button class="add-btn" onclick="addToCart(${p.id})">+ დამატება</button>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        function filterCat(cat, btn) {
            document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            renderProducts(cat);
        }

        function addToCart(id) {
            cart[id] = (cart[id] || 0) + 1;
            updateCart();
        }

        function changeQty(id, delta) {
            if (cart[id]) {
                cart[id] += delta;
                if (cart[id] <= 0) delete cart[id];
            }
            updateCart();
        }

        function updateCart() {
            let totalQty = 0;
            let totalPrice = 0;
            const container = document.getElementById('cart-items');
            container.innerHTML = '';

            Object.keys(cart).forEach(id => {
                const p = products.find(prod => prod.id == id);
                const qty = cart[id];
                totalQty += qty;
                totalPrice += p.price * qty;

                container.innerHTML += `
                    <div class="cart-item">
                        <div>
                            <strong>${p.name}</strong><br>
                            <small>${p.price} € x ${qty}</small>
                        </div>
                        <div class="cart-item-controls">
                            <button onclick="changeQty(${id}, -1)">-</button>
                            <span style="margin: 0 5px;">${qty}</span>
                            <button onclick="changeQty(${id}, 1)">+</button>
                        </div>
                    </div>
                `;
            });

            document.getElementById('cart-count').innerText = totalQty;
            document.getElementById('cart-total').innerText = totalPrice;
        }

        function toggleCart() {
            const m = document.getElementById('cart-modal');
            m.style.display = m.style.display === 'flex' ? 'none' : 'flex';
        }

        function handleCheckout(e) {
            e.preventDefault();
            if (Object.keys(cart).length === 0) {
                alert('კალათა ცარიელია!');
                return;
            }

            const name = document.getElementById('cust-name').value;
            const phone = document.getElementById('cust-phone').value;
            const addr = document.getElementById('cust-address').value;
            const date = document.getElementById('event-date').value;
            const guests = document.getElementById('guest-count').value;
            const pay = document.getElementById('pay-method').value;

            let text = `✨ *ახალი შეკვეთა CATERING-ზე* ✨

`;
            text += `👤 *სახელი:* ${name}
📞 *ტელ:* ${phone}
📍 *მისამართი:* ${addr}
📅 *თარიღი:* ${date}
👥 *სტუმრები:* ${guests}
💳 *გადახდა:* ${pay}

*შეკვეთილი პროდუქტები:*
`;

            let total = 0;
            Object.keys(cart).forEach(id => {
                const p = products.find(prod => prod.id == id);
                text += `• ${p.name} x ${cart[id]} (${p.price * cart[id]} €)
`;
                total += p.price * cart[id];
            });

            text += `
💰 *სულ ჯამი:* ${total} €`;

            const url = `https://wa.me/${MY_WHATSAPP_NUMBER}?text=${encodeURIComponent(text)}`;
            window.open(url, '_blank');
        }
